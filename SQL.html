Requête 1)c:


SELECT pds.NomProd, cat.NomCateg ,pds.CoutAchat
FROM (((produit AS pds
INNER JOIN Fournisseur AS fnsr ON pds.NoFour = fnsr.NoFour)INNER JOIN Pays ON Pays.NoPays = fnsr.NoPays)
INNER JOIN Categorie AS cat ON cat.CodeCateg = pds.CodeCateg)
WHERE (cat.NomCateg = "Viandes" OR cat.NomCateg LIKE "Poissons*") AND
(pds.CoutAchat >50)
ORDER BY cat.NomCateg, pds.CoutAchat, Pays.PaysEN

Requête 1)e:

SELECT Fournisseur.Societe, Categorie.NomCateg
FROM Fournisseur INNER JOIN
(Categorie SELECT
    Fournisseur.Societe
FROM
    Fournisseur
WHERE
    Fournisseur.NoFour IN (
        SELECT DISTINCT
            Fournisseur.NoFour
        FROM
            (
                Fournisseur
                INNER JOIN Produit ON Fournisseur.NoFour = Produit.NoFour
            )
            INNER JOIN Categorie ON Produit.CodeCateg = Categorie.CodeCateg
        WHERE
            Categorie.NomCateg IN ("Viandes")
    )
    AND Fournisseur.NoFour IN (
        SELECT DISTINCT
            Fournisseur.NoFour
        FROM
            (Fournisseur
                INNER JOIN Produit ON Fournisseur.NoFour = Produit.NoFour)
            INNER JOIN Categorie ON Produit.CodeCateg = Categorie.CodeCateg
        WHERE
            Categorie.NomCateg IN ("Poissons et fruits de mer")
    )
ORDER BY
    Fournisseur.Societe;

	ON Pays.NoPays = Fournisseur.NoPays)

	ON Fournisseur.NoFour = Produit.NoFour)

	ON Produit.RefProd = DetailCommande.[Réf produit]
