Requête 1)e:

SELECT Fournisseur.Societe, Categorie.NomCateg
FROM Fournisseur INNER JOIN
(Categorie SELECT
    Fournisseur.Societe
FROM
    Fournisseur
WHERE
    Fournisseur.NoFour IN (
        SELECT DISTINCT
            Fournisseur.NoFour
        FROM
            (
                Fournisseur
                INNER JOIN Produit ON Fournisseur.NoFour = Produit.NoFour
            )
            INNER JOIN Categorie ON Produit.CodeCateg = Categorie.CodeCateg
        WHERE
            Categorie.NomCateg IN ("Viandes")
    )
    AND Fournisseur.NoFour IN (
        SELECT DISTINCT
            Fournisseur.NoFour
        FROM
            (Fournisseur
                INNER JOIN Produit ON Fournisseur.NoFour = Produit.NoFour)
            INNER JOIN Categorie ON Produit.CodeCateg = Categorie.CodeCateg
        WHERE
            Categorie.NomCateg IN ("Poissons et fruits de mer")
    )
ORDER BY
    Fournisseur.Societe;

Le résultat :

Moines de Paris

Requête 2:

SELECT SUM (DetailCommande.Quantité * Produit.CoutAchat * (1 - DetailCommande.[Remise (%)] / 100))
FROM ((Pays INNER JOIN Fournisseur
	ON Pays.NoPays = Fournisseur.NoPays)
INNER JOIN Produit
	ON Fournisseur.NoFour = Produit.NoFour)
INNER JOIN DetailCommande
	ON Produit.RefProd = DetailCommande.[Réf produit]
WHERE Pays.UE = "Membre";

Requête 4

SELECT TOP 25
    pds.RefProd,
    pds.NomProd,
    pds.CoutAchat,
    dcmd.[Remise (%)],
    SUM(dcmd.Quantité) AS QuantiteVendue
FROM Produit AS pds
INNER JOIN DetailCommande AS dcmd ON pds.RefProd = dcmd.[Réf produit]
GROUP BY pds.RefProd, pds.NomProd, pds.CoutAchat, dcmd.[Remise (%)]
ORDER BY SUM(dcmd.Quantité) ASC, pds.CoutAchat DESC
